# Backend dependencies for the Enterprise AI Agent Platform.
# Offline-first: everything here runs and tests with zero external services
# and zero API keys.

# Web framework + server
fastapi
uvicorn[standard]

# Persistence
sqlalchemy
psycopg2-binary          # PostgreSQL driver (Docker only; SQLite needs nothing)

# Config + validation
pydantic
pydantic-settings
email-validator          # required by pydantic EmailStr

# Auth
bcrypt                   # password hashing (used directly; passlib is unmaintained)
python-jose[cryptography]  # JWT encode/decode

# Cache / session memory (optional; falls back to in-process store if absent)
redis

# --- dev / test only ---
pytest
httpx                    # FastAPI TestClient transport
